# Point-MAE

第一阶段：预训练 Point-MAE

cd /data/workspace/PCP-MAE

# 对 660K 个 Objaverse 样本进行纯 Point-MAE（无中心预测）预训练
# 单 GPU 训练，有效批次大小 64，300 epochs，bfloat16 AMP
python main.py \
    --config cfgs/pretrain/ablation_point_mae.yaml \
    --exp_name point_mae_objaverse \
    --seed 42

# 训练将日志记录到：
#   experiments/ablation_point_mae/pretrain/point_mae_objaverse/
# 检查点将保存为：
#   experiments/ablation_point_mae/pretrain/point_mae_objaverse/ckpt-best.pth
#   experiments/ablation_point_mae/pretrain/point_mae_objaverse/ckpt-last.pth

第二阶段：导出编码器权重
训练完成后，提取骨干权重以适配 MiniGPT-3D：


cd /data/workspace/PCP-MAE

# 从最佳检查点提取仅编码器权重
python ckpt_extract.py \
    --ckpt experiments/ablation_point_mae/pretrain/point_mae_objaverse/ckpt-last.pth \
    --out point_model_pointmae.pth
